<Quiz n={2} lesson="8.4" type="number" answer={13.5}>

<Quiz n={3} lesson="8.6" type="number" answer={9}>

<div slot="explain">$\tfrac{4}{6} = \tfrac{6}{EC}$، إذن $EC = 9$.</div>
</Quiz>
